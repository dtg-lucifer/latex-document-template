% prelude.tex
% Unified single-import entry point for Charlie's LaTeX Template.
% Automatically loads all required packages, configurations, and components.

\RequirePackage{iftex}

% Font and Typography Setup
\ifLuaTeX
    \RequirePackage{fontspec}
    \defaultfontfeatures{Ligatures=TeX}
    % Times New Roman (TeX Gyre Termes) everywhere for all regular text and headings
    \setmainfont{texgyretermes-regular.otf}[
        BoldFont = texgyretermes-bold.otf,
        ItalicFont = texgyretermes-italic.otf,
        BoldItalicFont = texgyretermes-bolditalic.otf,
    ]
    % Also set sans-serif font to Times New Roman so all text uses Times New Roman
    \setsansfont{texgyretermes-regular.otf}[
        BoldFont = texgyretermes-bold.otf,
        ItalicFont = texgyretermes-italic.otf,
        BoldItalicFont = texgyretermes-bolditalic.otf,
    ]
    % SF Mono for all monospaced / code typography and inline code
    \setmonofont{SFMono}[
        Path = assets/fonts/,
        Extension = .otf,
        UprightFont = * Regular Nerd Font Complete,
        BoldFont = * Bold Nerd Font Complete,
        ItalicFont = * Regular Italic Nerd Font Complete,
        BoldItalicFont = * Bold Italic Nerd Font Complete,
    ]
\else
    \RequirePackage[T1]{fontenc}
    \RequirePackage[utf8]{inputenc}
    \RequirePackage{newtxtext}
\fi
