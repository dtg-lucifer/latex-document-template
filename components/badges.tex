% components/badges.tex
% Status badges, MITRE ATT&CK tags, and metadata chips.

\RequirePackage[most]{tcolorbox}
\RequirePackage{xcolor}

% 1. Generic Pill Badge
\newcommand{\badge}[3][slate100]{%
    \tcbox[
        on line,
        boxrule=0.4pt,
        arc=3pt,
        colback=#1,
        colframe=slate300,
        left=5pt, right=5pt, top=1.5pt, bottom=1.5pt,
    ]{%
        \scriptsize\bfseries\sffamily\color{#2}#3%
    }%
}

% 2. System Policy & Status Badge
% Recognizes: ALLOW, NORMAL, ALERT, CRITICAL, ISOLATE, SUSPICIOUS, INFO, SUCCESS, DANGER
\newcommand{\statusbadge}[1]{%
    \lowercase{\def\tmpstatus{#1}}%
    \ifdefstring{\tmpstatus}{allow}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgegreenbg, colframe=badgegreenborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgegreentext}ALLOW}%
    }{%
    \ifdefstring{\tmpstatus}{success}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgegreenbg, colframe=badgegreenborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgegreentext}SUCCESS}%
    }{%
    \ifdefstring{\tmpstatus}{alert}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeamberbg, colframe=badgeamberborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeambertext}ALERT}%
    }{%
    \ifdefstring{\tmpstatus}{warning}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeamberbg, colframe=badgeamberborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeambertext}WARNING}%
    }{%
    \ifdefstring{\tmpstatus}{suspicious}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeamberbg, colframe=badgeamberborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeambertext}SUSPICIOUS}%
    }{%
    \ifdefstring{\tmpstatus}{critical}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeredbg, colframe=badgeredborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeredtext}CRITICAL}%
    }{%
    \ifdefstring{\tmpstatus}{isolate}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeredbg, colframe=badgeredborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeredtext}ISOLATE}%
    }{%
    \ifdefstring{\tmpstatus}{danger}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeredbg, colframe=badgeredborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeredtext}DANGER}%
    }{%
    \ifdefstring{\tmpstatus}{info}{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgeskybg, colframe=badgeskyborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgeskytext}INFO}%
    }{%
        \tcbox[on line, boxrule=0.4pt, arc=3pt, colback=badgenormalbg, colframe=badgenormalborder, left=5pt, right=5pt, top=1.5pt, bottom=1.5pt]{\scriptsize\bfseries\sffamily\color{badgenormaltext}#1}%
    }}}}}}}}}%
}

% 3. Tech / Component Tag
% Example: \techtag[PROTOCOL]{Raft Consensus}
\newcommand{\techtag}[2][]{%
    \tcbox[
        on line,
        boxrule=0.4pt,
        arc=3pt,
        colback=slate100,
        colframe=slate300,
        left=5pt, right=5pt, top=1.5pt, bottom=1.5pt,
    ]{%
        \ifstrempty{#1}{}{%
            {\tiny\bfseries\sffamily\color{accentsecondary}\textls[100]{\MakeUppercase{#1}}\ }%
        }%
        {\scriptsize\bfseries\ttfamily\color{accentprimary}#2}%
    }%
}

% 4. Monospace Protocol Chip
\newcommand{\protocoltag}[1]{%
    \tcbox[
        on line,
        boxrule=0.4pt,
        arc=3pt,
        colback=badgeskybg,
        colframe=badgeskyborder,
        left=4.5pt, right=4.5pt, top=1.2pt, bottom=1.2pt,
    ]{%
        {\scriptsize\bfseries\ttfamily\color{badgeskytext}#1}%
    }%
}

% 5. MITRE ATT&CK Technique Tag
% Example: \mitretag{T1046}[Network Service Scanning]
\newcommand{\mitretag}[2][]{%
    \tcbox[
        on line,
        boxrule=0.4pt,
        arc=3pt,
        colback=slate100,
        colframe=slate300,
        left=5pt, right=5pt, top=1.5pt, bottom=1.5pt,
    ]{%
        {\scriptsize\bfseries\ttfamily\color{accentprimary}#1}%
        \ifstrempty{#2}{}{%
            {\scriptsize\sffamily\color{slate500}: #2}%
        }%
    }%
}
