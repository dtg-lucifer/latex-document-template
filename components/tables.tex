\let\tablehead\relax
\newcommand{\tablehead}[1]{%
    \rowcolor{slate100}%
    \bfseries\sffamily\color{slate900}#1%
}

\newenvironment{styledtable}[1]{%
    \rowcolors{2}{slate50}{white}%
    \tabularx{\linewidth}{#1}%
}{%
    \endtabularx%
}

\newenvironment{styledtablenox}[1]{%
    \rowcolors{2}{slate50}{white}%
    \tabular{#1}%
}{%
    \endtabular%
}
