% components/cover.tex
% Minimalist Academic / Generic Document Cover Page.
% Modeled directly after the clean Typst math template cover page.
% Provides clean typography: centered title, subtitle, author, date, and bordered abstract.

\RequirePackage[most]{tcolorbox}
\RequirePackage{xcolor}
\RequirePackage{etoolbox}

\makeatletter

% Document Metadata Fields
\newcommand{\docsubtitle}[1]{\def\@docsubtitle{#1}}
\newcommand{\docorganization}[1]{\def\@docorganization{#1}}
\newcommand{\docversion}[1]{\def\@docversion{#1}}
\newcommand{\docdomain}[1]{\def\@docdomain{#1}}
\newcommand{\docabstract}[1]{\def\@docabstract{#1}}
\newcommand{\docfooternote}[1]{\def\@docfooternote{#1}}
\newcommand{\docauthors}[1]{\def\@docauthors{#1}}

% Defaults
\let\@docsubtitle\@empty
\let\@docorganization\@empty
\let\@docversion\@empty
\let\@docdomain\@empty
\let\@docabstract\@empty
\let\@docfooternote\@empty
\let\@docauthors\@empty

% Helper to print authors cleanly
\newcommand{\@printauthors}{%
    \ifdefempty{\@docauthors}{%
        \ifdefempty{\@author}{}{%
            \begingroup%
            \def\and{\quad\textbullet\quad}%
            \@author%
            \endgroup%
        }%
    }{\@docauthors}%
}

% Render Clean Cover Page
\newcommand{\rendercoverpage}{%
    \begin{titlepage}%
    \thispagestyle{empty}%
    \vspace*{\stretch{1}}%
    \begin{center}%
        \parbox{0.85\linewidth}{%
            \centering%
            {\fontsize{26}{32}\selectfont\bfseries\color{doc}\@title\par}%
            \ifdefempty{\@docsubtitle}{}{%
                \vspace{10pt}%
                {\fontsize{13}{18}\selectfont\itshape\color{slate500}\@docsubtitle\par}%
            }%
            \ifboolexpr{ test {\ifdefempty{\@docauthors}} and test {\ifdefempty{\@author}} }{}{%
                \vspace{18pt}%
                {\fontsize{13}{18}\selectfont\color{slate900}\@printauthors\par}%
            }%
            \ifdefempty{\@date}{}{%
                \vspace{8pt}%
                {\fontsize{10}{14}\selectfont\color{slate500}\@date\par}%
            }%
        }%
    \end{center}%
    \ifdefempty{\@docabstract}{%
        \vspace*{\stretch{2}}%
    }{%
        \vspace*{\stretch{2}}%
        \begin{center}%
        \begin{tcolorbox}[
            enhanced,
            blanker,
            width=0.80\linewidth,
            borderline north={0.5pt}{0pt}{slate300},
            borderline south={0.5pt}{0pt}{slate300},
            top=12pt,
            bottom=12pt,
            left=14pt,
            right=14pt,
            halign=left,
        ]%
            {\fontsize{9.5}{13}\selectfont\bfseries\color{doc} Abstract}\par\vspace{4pt}%
            {\fontsize{9}{13.5}\selectfont\itshape\color{slate900}\@docabstract\par}%
        \end{tcolorbox}%
        \end{center}%
        \vspace*{\stretch{2}}%
    }%
    \end{titlepage}%
    \clearpage%
}

\makeatother
