% template.tex
% Master document configuration: typography, geometry, headers/footers, and links.

\RequirePackage[a4paper, top=2.5cm, bottom=2.5cm, left=2.5cm, right=2.5cm, headheight=27pt]{geometry}
\RequirePackage{fancyhdr}
\RequirePackage{titlesec}
\RequirePackage{microtype}
\RequirePackage{caption}

%==================================================
% Hyperref Settings
%==================================================

\hypersetup{
    colorlinks=true,
    linkcolor=doc,
    citecolor=accentsecondary,
    urlcolor=accentprimary,
    filecolor=accentcrimson,
    pdfstartview={FitH},
    pdfpagemode=UseOutlines,
    bookmarksnumbered=true,
    pdfborder={0 0 0},
    plainpages=false,
    pdfpagelabels=true,
}

%==================================================
% Running Headers & Footers
%==================================================

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{slate300}\leaders\hrule height \headrulewidth\hfill}}
\renewcommand{\footrule}{\hbox to\headwidth{\color{slate300}\leaders\hrule height \footrulewidth\hfill}}

\fancyhead[R]{\footnotesize\color{slate700}\nouppercase{\rightmark}}
\fancyhead[L]{\footnotesize\bfseries\color{accentprimary}\nouppercase{\leftmark}}
\fancyfoot[R]{\footnotesize\bfseries\color{slate900}\thepage}
\fancyfoot[L]{\footnotesize\color{slate500}Nexus Architecture $\cdot$ Systems Engineering}

\fancypagestyle{plain}{
    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0.4pt}
    \renewcommand{\footrule}{\hbox to\headwidth{\color{slate300}\leaders\hrule height \footrulewidth\hfill}}
    \fancyfoot[C]{\footnotesize\bfseries\color{slate900}\thepage}
}

%==================================================
% Section & Chapter Title Typography
%==================================================

\titleformat{\chapter}[display]
{\normalfont\huge\bfseries\color{accentprimary}}
{\filleft\Huge\bfseries\color{accentsecondary!40}CHAPTER \thechapter}
{1ex}
{\titlerule[1.5pt]\vspace{1.5ex}\filright}
[\vspace{1ex}\textcolor{slate300}{\rule{\linewidth}{0.5pt}}]

% Unnumbered chapters (Contents, Outlines, etc.) have no decorative rules
\titleformat{name=\chapter,numberless}
{\normalfont\huge\bfseries\color{accentprimary}}
{}
{0pt}
{}

\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{15pt}

\titleformat{\section}
{\normalfont\Large\bfseries\color{accentprimary}}
{\thesection}{1em}{}

\titleformat{\subsection}
{\normalfont\large\bfseries\color{slate700}}
{\thesubsection}{1em}{}

\titleformat{\subsubsection}
{\normalfont\normalsize\bfseries\color{slate700}}
{\thesubsubsection}{1em}{}

% Captions
\captionsetup{
    font=small,
    labelfont={bf,color=accentprimary},
    skip=8pt,
}
