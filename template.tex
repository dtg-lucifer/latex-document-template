\ChRuleWidth{1.2pt}

% Charlie's Signature Header Banner for Unnumbered Chapters (TOC, LOF, LOT, Acronyms)
\makeatletter
\def\@makeschapterhead#1{%
    \vspace*{25\p@}%
    {\parindent \z@ \raggedright \normalfont
        \begin{tikzpicture}[remember picture, overlay]%
            \pgftext[right,x=\linewidth,y=0.2cm]{\color{doc!60}\Huge\scshape\bfseries #1}%
            \draw[fill=doc!60,draw=doc!60] (\linewidth-2.5cm,-.75) rectangle (\linewidth+4.5cm,1);%
            \clip (\linewidth-2.5cm,-.75) rectangle (\linewidth+4.5cm,1);%
            \pgftext[right,x=\linewidth,y=0.2cm]{\color{white}\Huge\scshape\bfseries #1}%
        \end{tikzpicture}\par
        \nobreak
        \vspace*{55\p@}%
    }%
}
\makeatother
